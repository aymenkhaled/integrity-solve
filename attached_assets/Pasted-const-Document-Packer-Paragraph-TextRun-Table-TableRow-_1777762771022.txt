// ── Helpers ──
const sp = (before=60, after=60) => ({ before, after });

function h1(text) {
  return new Paragraph({
    heading: HeadingLevel.HEADING_1,
    spacing: sp(400, 200),
    shading: { fill: NAVY, type: ShadingType.CLEAR },
    border: { bottom: { style: BorderStyle.SINGLE, size: 6, color: EMERALD } },
    indent: { left: 240, right: 240 },
    children: [new TextRun({ text, bold: true, size: 38, color: WHITE, font: "Arial" })]
  });
}

function h2(text) {
  return new Paragraph({
    heading: HeadingLevel.HEADING_2,
    spacing: sp(320, 140),
    border: { bottom: { style: BorderStyle.SINGLE, size: 3, color: EMERALD, space: 4 } },
    children: [new TextRun({ text, bold: true, size: 30, color: NAVY, font: "Arial" })]
  });
}

function h3(text) {
  return new Paragraph({
    heading: HeadingLevel.HEADING_3,
    spacing: sp(220, 80),
    children: [new TextRun({ text, bold: true, size: 24, color: SURFACE, font: "Arial" })]

}

function h4(text) {
  return new Paragraph({
    spacing: sp(180, 60),
    children: [new TextRun({ text, bold: true, size: 22, color: EMERALD, font: "Arial" })]

}

function p(text, opts = {}) {
  return new Paragraph({
    spacing: sp(60, 60),
    children: [new TextRun({ text, size: 21, font: "Arial", color: "1E293B", ...opts })]
  });
}

function pBold(text) {
  return p(text, { bold: true });
}

function bullet(text, level = 0) {
  return new Paragraph({
    numbering: { reference: "bullets", level },
    spacing: sp(40, 40),
    children: [new TextRun({ text, size: 20, font: "Arial", color: "334155" })]
  });
}

function num(text, level = 0) {
  return new Paragraph({
    numbering: { reference: "numbers", level },
    spacing: sp(40, 40),
    children: [new TextRun({ text, size: 20, font: "Arial", color: "334155" })]
  });
}

function code(text) {
  return new Paragraph({
    spacing: sp(30, 30),
    shading: { fill: CODE_BG, type: ShadingType.CLEAR },
    indent: { left: 360, right: 360 },
    children: [new TextRun({ text, size: 17, font: "Courier New", color: CODE_TXT })]
  });
}

function callout(text, type = "info") {
  const map = {
    info:    { fill: INFO_BG, color: INFO_CLR, prefix: "ℹ " },
    warn:    { fill: WARN_BG, color: WARN_CLR, prefix: "⚠ " },
    ok:      { fill: OK_BG,   color: OK_CLR,   prefix: "✅ " },
    danger:  { fill: ERR_BG,  color: ERR_CLR,  prefix: "🚨 " },
    diamond: { fill: "F5F3FF", color: "6D28D9", prefix: "💎 " },
    gap:     { fill: "FFF7ED", color: "9A3412", prefix: "🔧 GAP: " },
  };
  const { fill, color, prefix } = map[type] || map.info;
  return new Paragraph({
    spacing: sp(80, 80),
    shading: { fill, type: ShadingType.CLEAR },
    border: { left: { style: BorderStyle.SINGLE, size: 14, color: EMERALD, space: 4 } },
    indent: { left: 360, right: 240 },
    children: [new TextRun({ text: prefix + text, size: 20, font: "Arial", color, italics: true })]
  });
}

function spacer(n = 1) {
  return new Paragraph({ spacing: { before: 80 * n, after: 0 }, children: [new TextRun("")] });
}

function divider() {
  return new Paragraph({
    spacing: sp(120, 120),
    border: { bottom: { style: BorderStyle.SINGLE, size: 4, color: EMERALD } },
    children: [new TextRun("")]
  });
}

function table(headers, rows, colWidths, options = {}) {
  const totalW = colWidths.reduce((a, b) => a + b, 0);
  return new Table({
    width: { size: totalW, type: WidthType.DXA },
    columnWidths: colWidths,
    rows: [
      new TableRow({
        tableHeader: true,
        children: headers.map((h, i) => new TableCell({
          borders,
          width: { size: colWidths[i], type: WidthType.DXA },
          shading: { fill: options.headerBg || NAVY, type: ShadingType.CLEAR },
          margins: { top: 100, bottom: 100, left: 120, right: 120 },
          verticalAlign: VerticalAlign.CENTER,
          children: [new Paragraph({
            spacing: { before: 20, after: 20 },
            children: [new TextRun({ text: h, bold: true, size: 18, font: "Arial", color: WHITE })]
          })]
        }))
      }),
      ...rows.map((row, ri) => new TableRow({
        children: row.map((cell, ci) => {
          const cellText = String(cell ?? '');
          return new TableCell({
            borders,
            width: { size: colWidths[ci], type: WidthType.DXA },
            shading: { fill: ri % 2 === 0 ? WHITE : LIGHT_BG, type: ShadingType.CLEAR },
            margins: { top: 80, bottom: 80, left: 120, right: 120 },
            verticalAlign: VerticalAlign.TOP,
            children: [new Paragraph({
              spacing: { before: 20, after: 20 },
              children: [new TextRun({ text: cellText, size: 18, font: "Arial", color: "1E293B" })]
            })]
          });
        })
      }))
    ]
  });
}

function pageBreak() {
  return new Paragraph({ children: [new PageBreak()] });
}
